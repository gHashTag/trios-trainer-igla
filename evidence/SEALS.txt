# evidence/SEALS.txt - the published declaration seals, machine-readable.
#
# One line per tracked sidecar under evidence/, as
#     <record path><TAB>sha256:<seal>
# Lines beginning with '#' are comments; there are no other line shapes.
#
# The seal is defined in src/provenance_seal.rs and re-implemented in
# interop/triosckp_reader.py; the two must agree on every line below or the
# SPECIFICATION is wrong, not the record. The prose table in
# docs/PROVENANCE-BINDING.md section 4 renders from this file.
#
# A seal is NOT a signature. These digests are only worth the channel a reader
# obtained them from: an adversary holding both the record and this file simply
# recomputes them. What they buy is that a forgery becomes detectable by anyone
# who read them, instead of undetectable by everyone.
#
# Regenerate:
#   for j in $(git ls-files evidence | grep '\.json$'); do
#     printf '%s\t%s\n' "$j" \
#       "$(./target/release/ckpt_replay --record "$j" --provenance-seal \
#          | grep -o 'sha256:[0-9a-f]*' | head -1)"
#   done
#
# Minted 2026-08-06, after `sha256`, `bytes`, `seed`, `step` and
# `gf16_floor_every` entered the sealed set. Every digest published before that
# date is superseded.
